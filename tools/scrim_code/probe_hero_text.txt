variables {
    global:
        0: Probe_White
        1: Probe_Amber
        2: Probe_Green
}

rule ("Probe: Colours") {
    event {
        Ongoing - Global;
    }
    actions {
        Set Global Variable(Probe_White, Color(White));
        Set Global Variable(Probe_Amber, Custom Color(255, 155, 0, 255));
        Set Global Variable(Probe_Green, Custom Color(40, 205, 60, 255));
    }
}

    event {
        Ongoing - Global;
    }
    actions {

        "ROW 2 - the control. The SAME heroes forced through heroIcon().\nIf row 1 and row 2 look the same, a bare Hero is a glyph and the ban\nfeature needs a workshop glyph reference set (design section 2)."

        Create HUD Text(All Players(All Teams), Null, Custom String("BANS  : {0} | {1}", Hero(Sombra), Hero(Mauga)), Null, Left, 5, Null, Global.Probe_White, Null, Visible To, Visible Always);
    }
